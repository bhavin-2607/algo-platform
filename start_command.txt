# iTerm2 Tab 1 — Backend (FastAPI Server)
cd /Users/bhavinamesara/Projects/algo-platform/backend && export PATH="/opt/homebrew/bin:$PATH" && source .venv/bin/activate && uvicorn app.main:app --reload --host 0.0.0.0 --port 8000

# iTerm2 Tab 2 — Celery Worker (macOS requires solo pool & fork safety flag)
cd /Users/bhavinamesara/Projects/algo-platform/backend && export PATH="/opt/homebrew/bin:$PATH" && source .venv/bin/activate && OBJC_DISABLE_INITIALIZE_FORK_SAFETY=YES celery -A app.worker.celery_app worker --loglevel=info -P solo

# iTerm2 Tab 3 — Frontend (Vite React UI)
cd /Users/bhavinamesara/Projects/algo-platform/frontend && export PATH="/opt/homebrew/bin:$PATH" && npm run dev

# iTerm2 Tab 4 — Celery Beat (Periodic Scheduler)
cd /Users/bhavinamesara/Projects/algo-platform/backend && export PATH="/opt/homebrew/bin:$PATH" && source .venv/bin/activate && celery -A app.worker.celery_app beat --loglevel=info
